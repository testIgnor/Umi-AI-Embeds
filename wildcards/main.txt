[##]
Makes the prompt
[/##]
[function make_prompt]

[call preprocess]

[if double_prompt_enable]

[chance 15]
[chance 50]
[call make_double]
[/chance]
[else]
[chance 50]
[call make_split]
[/chance]
[else]
[call make_princess]
[/else]
[/else]
[/chance]
[else]
[call make_single]
[/else]

[/if]
[else]
[call make_single]
[/else]

[call set_model]

[/function]


[##]
Chooses a franchise from all wildcard files
[/##]
[function choose_franchise]
[set characterType]
characters
[/set]

[chance 15]
[set characterType]
myspecies
[/set]
[/chance]

[if characterType="characters"]
[set basepath]%BASE_DIR%/templates/wildcards/characters/[/set]
[/if]
[elif characterType="myspecies"]
[set basepath]%BASE_DIR%/templates/wildcards/species/[/set]
[/elif]
[set searchpath][get basepath]*.txt[/set]
[set franchise][choose][filelist "{get searchpath}" _basename _hide_ext][/choose][/set]

[set characterprompt]
<[get characterType _before=[ _after=]][get franchise _before=[ _after=]]>
[/set]

[/function]


[##]
Sets gender_prompt
[/##]
[function set_gender]
[set gender]female[/set]

[chance 15]
[set gender]
[choose]
male
futanari
[/choose]
[/set]
[/chance]

[switch gender]
    [case "female"]
        [set gender_short]1girl[/set]
        [set eff_gen]female[/set]
    [/case]
    [case "futanari"]
        [set gender_short]1futa[/set]
        [set eff_gen]futanari[/set]
        [call get_futa]
    [/case]
    [case "male"]
        [set gender_short]1boy[/set]
        [set eff_gen]futanari[/set]
        [set gender_suffix], (genderswap, gender morph:1.5), <[ezpenis]>[/set]
    [/case]
[/switch]

[/function]

[##]
Makes a 1character prompt
[/##]
[function make_single]

[call set_overrides_pre]
[call set_age gender="{get gender}"]

[set character_preset]NSFW, (solo:1.5), [get characterprompt], [get artist], [get ageprompt][get gender_suffix], <[EZEmotion]>, <[GazeFormat]>[chance 20], <[ezpreg]>[/chance][chance 30], <[Meta][Pose]>[/chance]**(person in background, people in background:2.0)**[/set]

[set prompt_suffix]
    [chance 0]
        [call get_preset eff_gen="{get eff_gen}"]
    [/chance]
    [else]
        [chance 100]
            <[imageformat]>, ([call wildcards/MyLocation]:1.3), [choose][call wildcards/myclothing][/choose]
        [/chance]
        [else]
            <[imageformat]>, ([call wildcards/MyLocation]:1.3), <[eznude][get eff_gen _before=[ _after=]]>
        [/else]
    [/else]
[/set]

[call set_overrides_post]

[set umistring][get character_preset], [get prompt_suffix][/set]

[set retrieved_string][Umi _verbose="{get debugging}" _debug="{get debugging}"][get umistring][/Umi][/set]

[set p_prompt][choose _case=0][get retrieved_string][/choose][/set]
[set n_prompt][choose _case=1][get retrieved_string][/choose][/set]

[/function]



[##]
Main function
[/##]
[function main_function]

[call make_prompt]

[set no_lora]
[char-count <]
[get p_prompt]
[/char-count]
[/set]

[while "no_lora > 10"]
[call make_prompt]

[set no_lora]
[char-count <]
[get p_prompt]
[/char-count]
[/set]
[/while]

[if "no_lora > 6"]
[set p_prompt][multiply-lora-by-percent 75][get p_prompt][/multiply-lora-by-percent][/set]
[/if]

[if "no_lora > 8"]
[set p_prompt][half-lora][get p_prompt][/half-lora][/set]
[/if]

[set p_prompt]
[strip-lora-on-model-mismatch][get p_prompt][/strip-lora-on-model-mismatch]
[/set]

[get p_style], [get p_prompt]

[set negative_prompt][get n_style], [get n_prompt][/set]

[/function]

[call main_function]
